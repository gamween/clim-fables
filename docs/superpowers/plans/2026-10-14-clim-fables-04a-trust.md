```ts
// Level 1 of the clim x Fables integration: your off-chain keeper reads clim's fee and composes it with its own.
// Self-contained (viem only): copy this file into the keeper. Every call is an eth_call; nothing is ever sent.
// Live read of the testnet desk: bun docs/integrate/keeper-read.ts
import { createPublicClient, http, type Address, type PublicClient } from "viem";
import { robinhoodTestnet } from "viem/chains";

/** clim on Robinhood Chain testnet (chain 46630), from shared/deployments/robinhood-testnet.json. */
export const CLIM_TESTNET = {
  policy: "0x5f140A5e17826CFcEeeD7cD3FcAe4D56c9e0C519",
  riskDesk: "0x63f543430982685387C83e4ec55E5a26fd1371Ef",
} as const satisfies Record<string, Address>;

/** The four ClimFeePolicy views this file reads (a subset of shared/abis/ClimFeePolicy.json, checked by a test). */
export const climFeePolicyReadAbi = [
  { type: "function", name: "quote", stateMutability: "view", inputs: [], outputs: [{ name: "fee", type: "uint24" }, { name: "mode", type: "uint8" }] },
  {
    type: "function",
    name: "quoteDirectional",
    stateMutability: "view",
    inputs: [],
    outputs: [{ name: "fee0For1", type: "uint24" }, { name: "fee1For0", type: "uint24" }, { name: "mode", type: "uint8" }],
  },
  { type: "function", name: "lastObservation", stateMutability: "view", inputs: [], outputs: [{ name: "tObs", type: "uint40" }, { name: "seq", type: "uint32" }] },
  { type: "function", name: "tauKillSec", stateMutability: "view", inputs: [], outputs: [{ name: "", type: "uint32" }] },
] as const;

/** normal: fresh desk; degraded: venues disagree (the fee is at least the policy's safe fee); blind: no report for more
 *  than tauKillSec (ClimPoker refuses to poke, and a keeper should ignore clim). */
export type ClimMode = "normal" | "degraded" | "blind";
const MODES: Readonly<Record<number, ClimMode>> = { 0: "normal", 1: "degraded", 2: "blind" };

export type ClimQuote = {
  /** quote(): one fee for both directions, in pips (1 pip = 0.0001 %, 100 pips = 1 bp). */
  fee: number;
  /** quoteDirectional(): zero-for-one pays currency0 for currency1. */
  fee0For1: number;
  fee1For0: number;
  mode: ClimMode;
  /** The desk's last observation (unix seconds) and report counter. */
  tObs: number;
  seq: number;
  /** blockTimestamp - tObs; negative by up to 30 s when a report carries a future tObs (the desk's skew limit). */
  ageSec: number;
  tauKillSec: number;
  /** Every value above was read at this block. */
  blockNumber: bigint;
  blockTimestamp: number;
};

export type ReadOptions = {
  policy?: Address;
  /** Read this many blocks behind the head, so a load-balanced RPC whose nodes lag a little still serves the block. */
  lagBlocks?: bigint;
};

/** Reads clim's quote, both directions, the observation and the kill delay, all at one block. */
export async function readClimQuote(client: PublicClient, o: ReadOptions = {}): Promise<ClimQuote> {
  const address = o.policy ?? CLIM_TESTNET.policy;
  const lag = o.lagBlocks ?? 3n;
  const head = await client.getBlockNumber();
  const blockNumber = head > lag ? head - lag : head;
  const read = { address, abi: climFeePolicyReadAbi, blockNumber } as const;
  const [block, [fee, mode], [fee0For1, fee1For0], [tObs, seq], tauKillSec] = await Promise.all([
    client.getBlock({ blockNumber }),
    client.readContract({ ...read, functionName: "quote" }),
    client.readContract({ ...read, functionName: "quoteDirectional" }),
    client.readContract({ ...read, functionName: "lastObservation" }),
    client.readContract({ ...read, functionName: "tauKillSec" }),
  ]);
  const named = MODES[mode];
  if (named === undefined) throw new Error(`unknown ClimFeePolicy mode ${mode}`);
  const blockTimestamp = Number(block.timestamp);
  return { fee, fee0For1, fee1For0, mode: named, tObs, seq, ageSec: blockTimestamp - tObs, tauKillSec, blockNumber, blockTimestamp };
}

/** clim counts while its desk is not blind and its observation is at most maxAgeSec old (default: the policy's
 *  tauKillSec, the same rule ClimPoker applies on-chain). */
export function climUsable(clim: ClimQuote, maxAgeSec: number = clim.tauKillSec): boolean {
  return clim.mode !== "blind" && clim.ageSec <= maxAgeSec;
}

export type Composed = { fee: number; source: "keeper" | "clim"; climUsed: boolean };

/** A one-fee hook (ETH/USDG today): max(your fee, clim's quote()) while clim is usable, your fee alone otherwise.
 *  Your event overrides stay on top: clim can raise the fee, never lower yours. */
export function composeFee(keeperFeePips: number, clim: ClimQuote, maxAgeSec?: number): Composed {
  if (!climUsable(clim, maxAgeSec)) return { fee: keeperFeePips, source: "keeper", climUsed: false };
  return clim.fee > keeperFeePips ? { fee: clim.fee, source: "clim", climUsed: true } : { fee: keeperFeePips, source: "keeper", climUsed: true };
}

export type ComposedDirectional = { fee0For1: Composed; fee1For0: Composed };

/** A per-direction hook: the same rule on each side, with quoteDirectional()'s per-side fees. */
```

```ts
export type Level2Batch = {
  generatedBy: string;
  params: Level2Params & { feePokerRole: number; climRunnerRole: number; climRunnerLabel: string };
  requiredCaller: string;
  calls: Call[];
  multicall: { to: Address; data: Hex };

function selector(name: string): Hex {
  return toFunctionSelector(getAbiItem({ abi: climPokerAbi, name }) as AbiFunction);
}

function call(description: string, functionName: string, args: Array<string | number | string[]>): Call {
  const item = getAbiItem({ abi: accessManagerAbi, name: functionName }) as AbiFunction;
  const signature = `${functionName}(${item.inputs.map((i) => i.type).join(",")})`;
  return { description, function: signature, args, data: encodeFunctionData({ abi: accessManagerAbi, functionName, args }) };
}

export function buildLevel2Batch(p: Level2Params): Level2Batch {
  const poker = getAddress(p.poker);
  const accessManager = getAddress(p.accessManager);
  const runners = p.runners.map((r) => getAddress(r));
  if (runners.length === 0) throw new Error("at least one CLIM_RUNNER account is needed");
  const keeperSel = [selector("postEvent"), selector("postEventAndPoke")];
  const runnerSel = [selector("poke")];
  const calls: Call[] = [
    call("FEE_POKER to ClimPoker: it may call the hook's pokeFee (needs POKER_ADMIN, role 5)", "grantRole", [FEE_POKER, poker, 0]),
    call("postEvent and postEventAndPoke on ClimPoker for FEE_POKER: the keeper posts its event overrides there (needs ADMIN, role 0)", "setTargetFunctionRole", [poker, keeperSel, FEE_POKER]),
    call(`label role ${CLIM_RUNNER} (needs ADMIN)`, "labelRole", [CLIM_RUNNER, CLIM_RUNNER_LABEL]),
    call(`poke on ClimPoker for ${CLIM_RUNNER_LABEL} (needs ADMIN)`, "setTargetFunctionRole", [poker, runnerSel, CLIM_RUNNER]),
    ...runners.map((r, i) => call(`${CLIM_RUNNER_LABEL} to ${i === 0 ? "clim's runner" : `account ${i}`} ${r} (needs ADMIN, the default admin of a new role)`, "grantRole", [CLIM_RUNNER, r, 0])),
  ];
  return {
    generatedBy: "docs/integrate/level2-batch.ts",
    params: { ...p, accessManager, poker, runners, feePokerRole: FEE_POKER, climRunnerRole: CLIM_RUNNER, climRunnerLabel: CLIM_RUNNER_LABEL },
    requiredCaller: "an account holding ADMIN (role 0) and POKER_ADMIN (role 5) with no execution delay, or send the grantRole(FEE_POKER) call from a POKER_ADMIN holder and the rest from an ADMIN holder",
    calls,
    multicall: { to: accessManager, data: encodeFunctionData({ abi: accessManagerAbi, functionName: "multicall", args: [calls.map((c) => c.data)] }) },
    revoke: {
      stop: call("revoke FEE_POKER from ClimPoker: every poke reverts in the hook from the next block (needs POKER_ADMIN)", "revokeRole", [FEE_POKER, poker]),
      cleanup: runners.map((r) => call(`revoke ${CLIM_RUNNER_LABEL} from ${r} (needs ADMIN)`, "revokeRole", [CLIM_RUNNER, r])),
    },
    expect: {
      functionRoles: [
        { function: "postEvent(uint24,uint24,uint40)", selector: keeperSel[0] as Hex, role: FEE_POKER },
        { function: "postEventAndPoke(uint24,uint24,uint40)", selector: keeperSel[1] as Hex, role: FEE_POKER },
        { function: "poke()", selector: runnerSel[0] as Hex, role: CLIM_RUNNER },

      members: [{ role: FEE_POKER, account: poker }, ...runners.map((account) => ({ role: CLIM_RUNNER, account }))],
    },

export async function checkOnChain(b: Level2Batch, rpcUrl?: string): Promise<string[]> {
  const client = createPublicClient({ chain: robinhoodTestnet, transport: http(rpcUrl) });
  const am = b.params.accessManager;
  const problems: string[] = [];
  for (const f of b.expect.functionRoles) {
    const role = await client.readContract({ address: am, abi: accessManagerReadAbi, functionName: "getTargetFunctionRole", args: [b.params.poker, f.selector] });
    if (Number(role) !== f.role) problems.push(`${f.function}: role ${role}, expected ${f.role}`);

  return problems;
```

```ts

export type Cadence = { n: number; spanSec: number; medianGapSec: number; p90GapSec: number; perDay: number };

/** Cadence of a series of times (unix seconds, ascending): the gaps' median and p90, and the measured daily rate
 *  ((n - 1) events over the span, scaled to 86,400 s), which stays right when the gaps alternate. */
export function cadence(times: readonly number[]): Cadence {
  if (times.length < 2) throw new RangeError("cadence needs at least two times");
  const gaps = times.slice(1).map((t, i) => t - (times[i] as number));
  if (gaps.some((g) => g < 0)) throw new RangeError("times must be ascending");
  const spanSec = (times[times.length - 1] as number) - (times[0] as number);
  if (spanSec === 0) throw new RangeError("cadence needs a span longer than 0 s");
  return {
    n: times.length,
    spanSec,
    medianGapSec: percentile(gaps, 50),
    p90GapSec: percentile(gaps, 90),
    perDay: Math.round(((times.length - 1) * 86_400) / spanSec),
  };
}
```

```ts
export const COST_BEGIN = "<!-- clim:cost:begin -->";
export const COST_END = "<!-- clim:cost:end -->";

export type GasEvidence = {
  generatedAt: string;
  chainId: number;
  deskReport: GasSummary | null;
  poke: GasSummary | null;
  keeperPoke: GasSummary | null;
  arbSwap: GasSummary | null;
  retailSwap: GasSummary | null;
  cadence?: { fromBlock: number; toBlock: number; deskReport: Cadence | null; poke: Cadence | null };
};

const ROWS: Array<[keyof Pick<GasEvidence, "deskReport" | "poke" | "keeperPoke" | "arbSwap" | "retailSwap">, string]> = [
  ["deskReport", "Desk report (CRE simulator, through the mock forwarder)"],
  ["poke", "`ClimPoker.poke()` on pool D"],
  ["keeperPoke", "Keeper-style `pokeFee` on pool K"],
  ["arbSwap", "Arbitrage swap"],
  ["retailSwap", "Retail swap"],
];

const int = (n: number): string => n.toLocaleString("en-US");

/** An ETH amount with no exponent and no trailing zeros (8.6297e-7 -> "0.00000086297"). */
export function formatEth(x: number): string {
  const s = x.toFixed(12).replace(/0+$/, "");
  return s.endsWith(".") ? `${s}0` : s;
}

export function renderCostSection(e: GasEvidence): string {
  const c = e.cadence;
  if (c === undefined || c.deskReport === null || c.poke === null) {
    throw new Error("docs/evidence/gas-robinhood-testnet.json has no cadence: run `cd bots && bun run live:gas` first");
  }
  if (e.deskReport === null || e.poke === null) throw new Error("the evidence file has no desk report or poke receipt");
  const lines = [
    `_Generated by \`cd bots && bun run live:cost-doc\` from \`docs/evidence/gas-robinhood-testnet.json\` (generated ${e.generatedAt}, chain ${e.chainId}). Do not edit by hand._`,
    "",
    "| Transaction | Receipts | Gas, median | Gas, p90 | L1 gas, median | Cost at the median gas price |",
    "|---|---|---|---|---|---|",
  ];
  for (const [key, label] of ROWS) {
    const s = e[key];
    lines.push(s === null ? `| ${label} | 0 | not measured | | | |` : `| ${label} | ${int(s.n)} | ${int(s.gasUsedMedian)} | ${int(s.gasUsedP90)} | ${int(s.gasUsedForL1Median)} | ${formatEth(s.costMedianEth)} ETH |`);
  }
  const gwei = e.deskReport.effectiveGasPriceMedianWei / 1e9;
  const perDayEth = e.deskReport.costMedianEth * c.deskReport.perDay + e.poke.costMedianEth * c.poke.perDay;
  lines.push(
    "",
    `Median effective gas price of the desk reports: ${gwei} gwei.`,
    "",
    `Cadence, read from the chain over blocks ${int(c.fromBlock)} to ${int(c.toBlock)} (outages included): ${int(c.deskReport.n)} desk reports, a median gap of ${c.deskReport.medianGapSec} s (p90 ${c.deskReport.p90GapSec} s), ${int(c.deskReport.perDay)} reports a day; ${int(c.poke.n)} pokes, ${int(c.poke.perDay)} a day.`,
    "",

  );
  return lines.join("\n");

/** Replaces what sits between the two markers (markers kept); throws if either is missing. */
export function replaceBetween(doc: string, body: string, begin: string = COST_BEGIN, end: string = COST_END): string {
  const a = doc.indexOf(begin);
  const b = doc.indexOf(end);
  if (a === -1 || b === -1 || b < a) throw new Error(`markers ${begin} ... ${end} not found`);
  return `${doc.slice(0, a + begin.length)}\n${body}\n${doc.slice(b)}`;
}
```

````markdown
How it composes:

- **max, never min.** `composeFee` returns the higher of the keeper's fee and clim's `quote()`. clim can raise the
  fee, never lower the keeper's, so the keeper's event overrides (CPI, FOMC) stay on top.
- **A blind desk is ignored.** When the desk has not reported for more than `tauKillSec` (180 s), `mode` is `blind`
  and the policy's quote jumps to its safe fee (30 bp); `composeFee` then returns the keeper's fee alone, the same
  rule `ClimPoker` applies on-chain (it refuses to poke). A stricter freshness is one argument:
  `composeFee(myFeePips, clim, 60)`.
- **A degraded desk counts.** `degraded` means the venues disagree by more than 25 bp; the policy then quotes at least
  its safe fee, and `composeFee` keeps it.
- **One fee or two.** ETH/USDG's hook takes one fee for both directions: use `composeFee` with `quote()`. The newer
  per-direction hooks take two: `composeDirectional(my0For1, my1For0, clim)` composes each side with
  `quoteDirectional()`.
- **Units.** Fees are in pips: 100 pips = 1 bp; Fables' ETH/USDG band is 100 to 6,000 pips with a 700-pip flat fee.
````

````markdown
## 8. Documented design choices

- **Last writer wins in the hook's one override slot.** Fables' hook keeps one override per pool. If the keeper keeps
  poking the hook directly while CLIM_RUNNER pokes through `ClimPoker`, each overwrites the other, and a clim poke can
  replace a higher keeper override with clim's lower fee: `test_ADirectKeeperPokeIsOverwrittenByTheNextClimPoke`
  (`Level2Batch.t.sol`). The intended setup routes the keeper's event overrides through `postEvent` /
  `postEventAndPoke`, where they compose as `max`: `test_AnEventPostedThroughThePokerSurvivesTheNextClimPoke`.
- **The policy's band sits inside the hook's band (deployment invariant).** The testnet policy's band is 700 to
  6,000 pips; Fables' ETH/USDG hook takes 100 to 6,000 (spec §2). If Fables lowers the cap or raises the floor past
  the policy's, pokes outside the new band revert (section 7); redeploy the policy with the new band.
- **Desk owner: timelock or renounce.** Section 2. Until then, the owner key is trusted with the forwarder and the
  workflow choice.
- **Blind is not degraded.** Blind (no report for more than 180 s) means no poke at all and a return to flat when the
  last override expires; degraded (venues disagree beyond 25 bp) still pokes, at least the safe fee (30 bp).
- **A one-fee hook gets the higher side.** A symmetric `ClimPoker` sends the higher of the two directional fees to
  both directions (`test_SymmetricPokeSendsTheHigherSideToTheOldHook`), so a directional premium would raise the fee
  both ways. Keep `dirSlopeE4 = 0` for ETH/USDG until it moves to a per-direction hook; the recorded fork run does
  (`contracts/script/v2/ForkRun.s.sol` forces both directional parameters to 0). Pool D on testnet is per-direction
  (`ClimPoker.directional()` is `true`), so the premium is live there.
- **A future `tObs` up to 30 s** extends the desk's freshness window by as much; the override sent to the hook is
  still capped at `ttl` (section 7).
- **The keeper key can poke any pool on clim's testnet AccessManager** (section 5).
- **Simulation trusts one EOA** (section 6).
````

````markdown
## What it is

Fables' ETH/USDG pool on Robinhood Chain charges a flat fee, and Fables' keeper replaces it with an override posted
through the hook's `pokeFee`. The keeper is good: at equal average fee the lab finds clim's rule and the keeper
comparable, and the results below say where the keeper does better. clim does not offer better fees. It offers:

1. **Verifiability.** A public baseline recomputed every 30 s from four exchanges (Coinbase, Kraken, Binance,

## Results, as measured

Lab studies on Fables' ETH/USDG pool, 2026-09-07 to 2026-10-12, quoted from the `framing` field of each output:

- Effective arbitrage interval ([`lab/out/fables-dt-eff.json`](lab/out/fables-dt-eff.json)): "dt_eff is of order 1 to 3 s; the per-hour estimate is very sensitive to the band (a 1 bp shift in the reference price moves weeks 4-5 from ~13 s to ~3-4.5 s). clim's posted fee does not depend on dt: eta is calibrated at equal average fee, so only eta*sqrt(dt/2) matters."
- Volatility window ([`lab/out/fables-window.json`](lab/out/fables-window.json)): "At equal time-average fee, clim's 5-minute rule and the keeper are comparable: -4.4 % arbitrage loss with arbitrage every second, +4.6 % every 15 s; clim charges 11.96 bp per dollar traded vs the keeper's 10.77 bp. Not a claim of better fees."
- Volatility estimator ([`lab/out/fables-estimator.json`](lab/out/fables-estimator.json)): "binance-1s scored best in-sample by 1.09 points on a simulator driven by the same Binance path; it is excluded because the desk must not depend on one venue (spec §3); venue-range is the best multi-venue estimator."
- Per-direction signal ([`lab/out/fables-direction.json`](lab/out/fables-direction.json)): "Trades toward the venue capture the stale pool's gap (signal correlates 0.93 with the pre-trade gap and 0.04 with the later move); the premium rule clears its gate on the five weeks pooled but in only 3 of 5 weeks; measured, experimental, not deployed on ETH/USDG (its hook is symmetric)."

The decided parameters are in [`shared/params-fables.json`](shared/params-fables.json). The studies rerun from
[`lab/README.md`](lab/README.md) (the raw data is re-downloadable with the fetch scripts; the swap history comes
from a separate download the README names).
````
